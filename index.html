const SHEET_NAME = "Orders"; /* ===================================================== SLS STORE - GOOGLE SHEETS ORDER SYSTEM ===================================================== Delivery: Inside Dhaka = 80 BDT Outside Dhaka = 150 BDT ===================================================== */ function setupSLSStore() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } const headers = [ "Order ID", "Date & Time", "Customer Name", "Phone", "Address", "Delivery Area", "Products", "Products Total (BDT)", "Delivery Charge (BDT)", "Grand Total (BDT)", "Payment Method", "Status" ]; /* যদি Sheet একদম খালি থাকে তাহলে Header তৈরি করবে */ if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.setFrozenRows(1); } /* Header format */ sheet .getRange(1, 1, 1, headers.length) .setFontWeight("bold"); sheet.autoResizeColumns( 1, headers.length ); } /* ===================================================== GET REQUEST ===================================================== */ function doGet(e) { return ContentService .createTextOutput( JSON.stringify({ success: true, message: "SLS Store Order System is working!" }) ) .setMimeType( ContentService.MimeType.JSON ); } /* ===================================================== POST ORDER ===================================================== */ function doPost(e) { try { setupSLSStore(); /* Check data */ if ( !e || !e.postData || !e.postData.contents ) { return jsonResponse({ success: false, message: "No order data received." }); } /* Read order data */ const data = JSON.parse( e.postData.contents ); /* Basic validation */ if ( !data.name || !data.phone || !data.address || !data.items || !Array.isArray(data.items) || data.items.length === 0 ) { return jsonResponse({ success: false, message: "Customer information or products are missing." }); } /* Delivery Area */ const deliveryArea = data.deliveryArea === "Inside Dhaka" ? "Inside Dhaka" : "Outside Dhaka"; /* Delivery Charge */ const deliveryCharge = deliveryArea === "Inside Dhaka" ? 80 : 150; /* Product Total */ const productTotal = data.items.reduce( function(sum, item) { return ( sum + (Number(item.price) || 0) * (Number(item.qty) || 0) ); }, 0 ); /* Grand Total */ const grandTotal = productTotal + deliveryCharge; /* ================================================= DUPLICATE ORDER PROTECTION ================================================= */ const clientOrderId = String( data.clientOrderId || "" ).trim(); const properties = PropertiesService .getScriptProperties(); if (clientOrderId) { const oldOrderId = properties.getProperty( "CLIENT_ORDER_" + clientOrderId ); /* যদি একই Order আবার আসে */ if (oldOrderId) { return jsonResponse({ success: true, duplicate: true, orderId: oldOrderId, message: "This order has already been received." }); } } /* ================================================= CREATE ORDER ID ================================================= */ const orderId = createOrderId(); /* ================================================= PRODUCT LIST ================================================= */ const products = data.items .map(function(item) { return ( item.name + " x" + item.qty + " @ ৳" + item.price ); }) .join(" | "); /* ================================================= SAVE TO GOOGLE SHEET ================================================= */ const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); sheet.appendRow([ orderId, new Date(), String(data.name).trim(), String(data.phone).trim(), String(data.address).trim(), deliveryArea, products, productTotal, deliveryCharge, grandTotal, "Cash on Delivery", "Pending" ]); /* ================================================= SAVE CLIENT ORDER ID ================================================= */ if (clientOrderId) { properties.setProperty( "CLIENT_ORDER_" + clientOrderId, orderId ); } /* ================================================= SUCCESS ================================================= */ return jsonResponse({ success: true, orderId: orderId, productTotal: productTotal, deliveryCharge: deliveryCharge, grandTotal: grandTotal, message: "Order received successfully." }); } catch (error) { return jsonResponse({ success: false, message: String(error) }); } } /* ===================================================== CREATE ORDER ID ===================================================== */ function createOrderId() { const properties = PropertiesService .getScriptProperties(); let number = Number( properties.getProperty( "ORDER_NUMBER" ) || "0" ); number++; properties.setProperty( "ORDER_NUMBER", String(number) ); return ( "SLS-" + String(number) .padStart(5, "0") ); } /* ===================================================== JSON RESPONSE ===================================================== */ function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
