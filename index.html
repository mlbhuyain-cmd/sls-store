const SHEET_NAME = "Orders"; // ===================================================== // SLS STORE - GOOGLE SHEETS ORDER SYSTEM // ===================================================== // Product Total = শুধু পণ্যের মোট দাম // Delivery Charge = Inside Dhaka ৳80 / Outside Dhaka ৳150 // Grand Total = Product Total + Delivery Charge // ===================================================== // ----------------------------------------------------- // SETUP SHEET // ----------------------------------------------------- function setupSLSStore() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } const headers = [ "Order ID", "Date & Time", "Customer Name", "Phone", "Address", "Delivery Area", "Products", "Product Total (BDT)", "Delivery Charge (BDT)", "Grand Total (BDT)", "Payment Method", "Status" ]; // যদি sheet একদম খালি থাকে if (sheet.getLastRow() === 0) { sheet.getRange(1, 1, 1, headers.length) .setValues([headers]); } else { // প্রথম row-তে নতুন header বসানো sheet.getRange(1, 1, 1, headers.length) .setValues([headers]); } // Header formatting sheet.getRange(1, 1, 1, headers.length) .setFontWeight("bold"); sheet.setFrozenRows(1); // Number formatting if (sheet.getMaxRows() > 1) { sheet.getRange(2, 8, sheet.getMaxRows() - 1, 3) .setNumberFormat('৳#,##0'); } sheet.autoResizeColumns(1, headers.length); } // ----------------------------------------------------- // GET REQUEST TEST // ----------------------------------------------------- function doGet(e) { return ContentService .createTextOutput( JSON.stringify({ success: true, message: "SLS Store Order System is working!" }) ) .setMimeType(ContentService.MimeType.JSON); } // ----------------------------------------------------- // RECEIVE ORDER // ----------------------------------------------------- function doPost(e) { try { setupSLSStore(); // Check incoming data if (!e || !e.postData || !e.postData.contents) { return jsonResponse({ success: false, message: "No order data received." }); } // Convert JSON data const data = JSON.parse(e.postData.contents); // ------------------------------------------------- // BASIC VALIDATION // ------------------------------------------------- if ( !data.name || !data.phone || !data.address || !data.items || !Array.isArray(data.items) || data.items.length === 0 ) { return jsonResponse({ success: false, message: "Customer information or products are missing." }); } // ------------------------------------------------- // DELIVERY AREA // ------------------------------------------------- const deliveryArea = data.deliveryArea === "Inside Dhaka" ? "Inside Dhaka" : "Outside Dhaka"; // ------------------------------------------------- // DELIVERY CHARGE // ------------------------------------------------- const deliveryCharge = deliveryArea === "Inside Dhaka" ? 80 : 150; // ------------------------------------------------- // PRODUCT TOTAL // ------------------------------------------------- const productTotal = data.items.reduce( function(sum, item) { const price = Number(item.price) || 0; const qty = Number(item.qty) || 0; return sum + (price * qty); }, 0 ); // ------------------------------------------------- // GRAND TOTAL // ------------------------------------------------- const grandTotal = productTotal + deliveryCharge; // ------------------------------------------------- // DUPLICATE ORDER PROTECTION // ------------------------------------------------- const clientOrderId = String(data.clientOrderId || "").trim(); const properties = PropertiesService.getScriptProperties(); if (clientOrderId) { const oldOrderId = properties.getProperty( "CLIENT_ORDER_" + clientOrderId ); if (oldOrderId) { return jsonResponse({ success: true, duplicate: true, orderId: oldOrderId, message: "This order has already been received." }); } } // ------------------------------------------------- // CREATE ORDER ID // ------------------------------------------------- const orderId = createOrderId(); // ------------------------------------------------- // PRODUCT LIST // ------------------------------------------------- const products = data.items .map(function(item) { return ( item.name + " x" + item.qty + " @ ৳" + item.price ); }) .join(" | "); // ------------------------------------------------- // GET SHEET // ------------------------------------------------- const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); // ------------------------------------------------- // SAVE ORDER // ------------------------------------------------- sheet.appendRow([ orderId, new Date(), String(data.name).trim(), String(data.phone).trim(), String(data.address).trim(), deliveryArea, products, productTotal, deliveryCharge, grandTotal, "Cash on Delivery", "Pending" ]); // ------------------------------------------------- // SAVE CLIENT ORDER ID // ------------------------------------------------- if (clientOrderId) { properties.setProperty( "CLIENT_ORDER_" + clientOrderId, orderId ); } // ------------------------------------------------- // FORMAT MONEY CELLS // ------------------------------------------------- const lastRow = sheet.getLastRow(); sheet .getRange(lastRow, 8, 1, 3) .setNumberFormat('৳#,##0'); // ------------------------------------------------- // SUCCESS RESPONSE // ------------------------------------------------- return jsonResponse({ success: true, orderId: orderId, productTotal: productTotal, deliveryCharge: deliveryCharge, grandTotal: grandTotal, message: "Order received successfully." }); } catch (error) { return jsonResponse({ success: false, message: String(error) }); } } // ----------------------------------------------------- // CREATE ORDER ID // ----------------------------------------------------- function createOrderId() { const properties = PropertiesService.getScriptProperties(); let number = Number( properties.getProperty("ORDER_NUMBER") || "0" ); number++; properties.setProperty( "ORDER_NUMBER", String(number) ); return ( "SLS-" + String(number).padStart(5, "0") ); } // ----------------------------------------------------- // JSON RESPONSE // ----------------------------------------------------- function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
