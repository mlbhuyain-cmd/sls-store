const SHEET_NAME = "Orders"; /* ======================================================== SLS STORE - GOOGLE SHEETS ORDER SYSTEM ======================================================== একজন Customer-এর পুরো Cart = ১টি Order Inside Dhaka Delivery = ৳80 Outside Dhaka Delivery = ৳150 ======================================================== */ function setupSLSStore() { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(SHEET_NAME); if (!sheet) { sheet = ss.insertSheet(SHEET_NAME); } const headers = [ "Order ID", "Date & Time", "Customer Name", "Phone", "Address", "Delivery Area", "Products", "Product Total (BDT)", "Delivery Charge (BDT)", "Grand Total (BDT)", "Payment Method", "Status" ]; /* যদি Sheet একদম খালি থাকে, তাহলে Header তৈরি করবে */ if (sheet.getLastRow() === 0) { sheet.appendRow(headers); sheet.setFrozenRows(1); sheet .getRange(1, 1, 1, headers.length) .setFontWeight("bold"); sheet.autoResizeColumns(1, headers.length); } } /* ======================================================== GET REQUEST ======================================================== */ function doGet(e) { return ContentService .createTextOutput( JSON.stringify({ success: true, message: "SLS Store Order System is working!" }) ) .setMimeType(ContentService.MimeType.JSON); } /* ======================================================== POST ORDER ======================================================== */ function doPost(e) { try { setupSLSStore(); /* ---------------------------------------------------- Check incoming data ---------------------------------------------------- */ if ( !e || !e.postData || !e.postData.contents ) { return jsonResponse({ success: false, message: "No order data received." }); } /* ---------------------------------------------------- Read customer order ---------------------------------------------------- */ const data = JSON.parse( e.postData.contents ); /* ---------------------------------------------------- Basic validation ---------------------------------------------------- */ if ( !data.name || !data.phone || !data.address || !data.items || !Array.isArray(data.items) || data.items.length === 0 ) { return jsonResponse({ success: false, message: "Customer information or products are missing." }); } /* ---------------------------------------------------- DELIVERY AREA ---------------------------------------------------- */ let deliveryArea = "Outside Dhaka"; if ( String(data.deliveryArea || "") .toLowerCase() .includes("inside") ) { deliveryArea = "Inside Dhaka"; } /* ---------------------------------------------------- DELIVERY CHARGE ---------------------------------------------------- */ let deliveryCharge = 150; if (deliveryArea === "Inside Dhaka") { deliveryCharge = 80; } /* ---------------------------------------------------- PRODUCT TOTAL ---------------------------------------------------- */ let productTotal = 0; data.items.forEach(function(item) { const price = Number(item.price) || 0; const qty = Number(item.qty) || 0; productTotal += price * qty; }); /* ---------------------------------------------------- GRAND TOTAL ---------------------------------------------------- */ const grandTotal = productTotal + deliveryCharge; /* ---------------------------------------------------- CREATE ONE ORDER ID ---------------------------------------------------- IMPORTANT: পুরো Cart-এর জন্য মাত্র ১টি Order ID হবে। Example: SLS-00001 Products: Lipstick x1 Serum x2 Foundation x1 সবগুলো একই Order ID-এর মধ্যে থাকবে। ---------------------------------------------------- */ const orderId = createOrderId(); /* ---------------------------------------------------- PRODUCT LIST ---------------------------------------------------- */ const products = data.items .map(function(item) { const name = String(item.name || ""); const qty = Number(item.qty) || 0; const price = Number(item.price) || 0; return ( name + " x" + qty + " @ ৳" + price ); }) .join(" | "); /* ---------------------------------------------------- GET SHEET ---------------------------------------------------- */ const sheet = SpreadsheetApp .getActiveSpreadsheet() .getSheetByName(SHEET_NAME); /* ---------------------------------------------------- SAVE ONE COMPLETE ORDER ---------------------------------------------------- */ sheet.appendRow([ orderId, new Date(), String(data.name).trim(), String(data.phone).trim(), String(data.address).trim(), deliveryArea, products, productTotal, deliveryCharge, grandTotal, "Cash on Delivery", "Pending" ]); /* ---------------------------------------------------- SUCCESS RESPONSE ---------------------------------------------------- */ return jsonResponse({ success: true, orderId: orderId, productTotal: productTotal, deliveryCharge: deliveryCharge, grandTotal: grandTotal, deliveryArea: deliveryArea, message: "Order received successfully." }); } catch (error) { return jsonResponse({ success: false, message: String(error) }); } } /* ======================================================== CREATE ORDER ID ======================================================== */ function createOrderId() { const properties = PropertiesService .getScriptProperties(); let number = Number( properties.getProperty( "ORDER_NUMBER" ) || "0" ); number++; properties.setProperty( "ORDER_NUMBER", String(number) ); return ( "SLS-" + String(number).padStart(5, "0") ); } /* ======================================================== JSON RESPONSE ======================================================== */ function jsonResponse(data) { return ContentService .createTextOutput( JSON.stringify(data) ) .setMimeType( ContentService.MimeType.JSON ); }
